CREATE TYPE hari_libur_jenis AS ENUM ('nasional', 'cuti_bersama', 'sekolah');
CREATE TYPE hari_libur_sumber AS ENUM ('manual', 'google_calendar');

CREATE TABLE hari_libur (
  id SERIAL PRIMARY KEY,
  tanggal DATE NOT NULL,
  nama VARCHAR(200) NOT NULL,
  deskripsi TEXT DEFAULT NULL,
  jenis hari_libur_jenis NOT NULL DEFAULT 'nasional',
  sumber hari_libur_sumber NOT NULL DEFAULT 'manual',
  gcal_id VARCHAR(255) DEFAULT NULL,
  gcal_link VARCHAR(500) DEFAULT NULL,
  aktif BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE (tanggal, nama),
  UNIQUE (gcal_id)
);

COMMENT ON COLUMN hari_libur.gcal_id IS 'Event ID dari Google Calendar';

CREATE INDEX idx_hari_libur_tanggal ON hari_libur (tanggal);
CREATE INDEX idx_hari_libur_jenis ON hari_libur (jenis);
CREATE INDEX idx_hari_libur_sumber ON hari_libur (sumber);

-- Trigger untuk auto-update updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = CURRENT_TIMESTAMP;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_hari_libur_updated_at
BEFORE UPDATE ON hari_libur
FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
